document.addEventListener('click', async (e) => { if (!currentTarget || !studentName || !e.target.matches('td[data-value]')) return; const isCorrect = (e.target.id === currentTarget); const status = isCorrect ? 'correct' : 'incorrect'; try { const res = await fetch(`${BASE_URL}/classes/GameRound?where={"targetCellId":"${encodeURIComponent(currentTarget)}"}&limit=1`, { headers: { 'X-LC-Id': APP_ID, 'X-LC-Key': APP_KEY } }); const data = await res.json(); if (data.results.length > 0) { const obj = data.results[0]; const attempts = { ...(obj.attempts || {}) }; attempts[studentName] = status; // record attempt // Also maintain completedBy for backward compatibility const completedBy = isCorrect ? [...new Set([...(obj.completedBy || []), studentName])] : obj.completedBy || []; await fetch(`${BASE_URL}/classes/GameRound/${obj.objectId}`, { method: 'PUT', headers: { 'X-LC-Id': APP_ID, 'X-LC-Key': APP_KEY, 'Content-Type': 'application/json' }, body: JSON.stringify({ attempts, completedBy }) }); if (isCorrect) { e.target.classList.add('correct'); } } } catch (err) { console.error('Submit error:', err); } });
